# Study Timetable Planner - Complete Mini Project This document contains the complete, ready-to-run source code and setup guide for the **Study Timetable Planner** mini-project built using Python Flask, HTML5, CSS3, JavaScript, and SQLite. --- ## 📁 Project Directory Structure Create a folder named `study_planner` and structure your project files as follows: ```text study_planner/ │ ├── app.py ├── schema.sql ├── static/ │ ├── css/ │ │ └── style.css │ └── js/ │ └── app.js └── templates/ └── index.html ``` --- ## 1. Database Schema (`schema.sql`) ```sql CREATE TABLE IF NOT EXISTS subjects ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL UNIQUE, duration INTEGER NOT NULL DEFAULT 60, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE IF NOT EXISTS timetable ( id INTEGER PRIMARY KEY AUTOINCREMENT, day TEXT NOT NULL, subject_id INTEGER NOT NULL, start_time TEXT NOT NULL, end_time TEXT NOT NULL, is_completed INTEGER DEFAULT 0, FOREIGN KEY (subject_id) REFERENCES subjects (id) ON DELETE CASCADE ); ``` --- ## 2. Flask Backend Application (`app.py`) ```python import sqlite3 from flask import Flask, render_template, request, jsonify app = Flask(__name__) DATABASE = 'timetable_planner.db' def get_db(): conn = sqlite3.connect(DATABASE) conn.row_factory = sqlite3.Row return conn def init_db(): with app.app_context(): db = get_db() with open('schema.sql', mode='r') as f: db.cursor().executescript(f.read()) db.commit() # --- Page Route --- @app.route('/') def index(): return render_template('index.html') # --- Dashboard API --- @app.route('/api/dashboard', methods=['GET']) def get_dashboard_stats(): db = get_db() total_subjects = db.execute("SELECT COUNT(*) FROM subjects").fetchone()[0] total_sessions = db.execute("SELECT COUNT(*) FROM timetable").fetchone()[0] completed_sessions = db.execute("SELECT COUNT(*) FROM timetable WHERE is_completed = 1").fetchone()[0] return jsonify({ "total_subjects": total_subjects, "total_sessions": total_sessions, "completed_sessions": completed_sessions }) # --- Subject APIs --- @app.route('/api/subjects', methods=['GET', 'POST']) def manage_subjects(): db = get_db() if request.method == 'GET': subjects = db.execute("SELECT * FROM subjects ORDER BY name ASC").fetchall() return jsonify([dict(s) for s in subjects]) elif request.method == 'POST': data = request.get_json() name = data.get('name', '').strip() duration = data.get('duration', 60) if not name: return jsonify({"error": "Subject name is required."}), 400 try: db.execute("INSERT INTO subjects (name, duration) VALUES (?, ?)", (name, duration)) db.commit() return jsonify({"message": "Subject added successfully!"}), 201 except sqlite3.IntegrityError: return jsonify({"error": "Subject already exists."}), 400 @app.route('/api/subjects/', methods=['DELETE']) def delete_subject(subject_id): db = get_db() db.execute("DELETE FROM subjects WHERE id = ?", (subject_id,)) db.commit() return jsonify({"message": "Subject deleted successfully!"}) # --- Timetable APIs --- @app.route('/api/timetable', methods=['GET', 'POST']) def manage_timetable(): db = get_db() if request.method == 'GET': query = """ SELECT t.id, t.day, t.start_time, t.end_time, t.is_completed, t.subject_id, s.name as subject_name FROM timetable t JOIN subjects s ON t.subject_id = s.id ORDER BY CASE WHEN t.day = 'Monday' THEN 1 WHEN t.day = 'Tuesday' THEN 2 WHEN t.day = 'Wednesday' THEN 3 WHEN t.day = 'Thursday' THEN 4 WHEN t.day = 'Friday' THEN 5 WHEN t.day = 'Saturday' THEN 6 WHEN t.day = 'Sunday' THEN 7 END, t.start_time ASC """ entries = db.execute(query).fetchall() return jsonify([dict(e) for e in entries]) elif request.method == 'POST': data = request.get_json() day = data.get('day') subject_id = data.get('subject_id') start_time = data.get('start_time') end_time = data.get('end_time') if not all([day, subject_id, start_time, end_time]): return jsonify({"error": "All fields are required."}), 400 if start_time >= end_time: return jsonify({"error": "End time must be later than start time."}), 400 db.execute( "INSERT INTO timetable (day, subject_id, start_time, end_time) VALUES (?, ?, ?, ?)", (day, subject_id, start_time, end_time) ) db.commit() return jsonify({"message": "Session scheduled successfully!"}), 201 @app.route('/api/timetable/', methods=['PUT', 'DELETE']) def update_delete_timetable(entry_id): db = get_db() if request.method == 'DELETE': db.execute("DELETE FROM timetable WHERE id = ?", (entry_id,)) db.commit() return jsonify({"message": "Schedule entry deleted!"}) elif request.method == 'PUT': data = request.get_json() if 'is_completed' in data: db.execute("UPDATE timetable SET is_completed = ? WHERE id = ?", (int(data['is_completed']), entry_id)) else: day = data.get('day') subject_id = data.get('subject_id') start_time = data.get('start_time') end_time = data.get('end_time') if start_time >= end_time: return jsonify({"error": "End time must be later than start time."}), 400 db.execute( "UPDATE timetable SET day = ?, subject_id = ?, start_time = ?, end_time = ? WHERE id = ?", (day, subject_id, start_time, end_time, entry_id) ) db.commit() return jsonify({"message": "Entry updated successfully!"}) if __name__ == '__main__': init_db() app.run(debug=True, port=5000) ``` --- ## 3. Client Logic (`static/js/app.js`) ```javascript document.addEventListener('DOMContentLoaded', () => { // API Endpoints const API_DASHBOARD = '/api/dashboard'; const API_SUBJECTS = '/api/subjects'; const API_TIMETABLE = '/api/timetable'; // DOM Elements const subjectForm = document.getElementById('subject-form'); const scheduleForm = document.getElementById('schedule-form'); const subjectList = document.getElementById('subject-list'); const subjectSelect = document.getElementById('sched-subject'); const timetableRows = document.getElementById('timetable-rows'); const editIdInput = document.getElementById('edit-id'); const submitSchedBtn = document.getElementById('submit-sched-btn'); const cancelEditBtn = document.getElementById('cancel-edit-btn'); const toast = document.getElementById('toast'); // Initial Data Fetch refreshAll(); function refreshAll() { fetchDashboardStats(); fetchSubjects(); fetchTimetable(); } // --- Toast Notification Helper --- function showToast(message, type = 'success') { toast.textContent = message; toast.className = `toast ${type}`; setTimeout(() => { toast.className = 'toast hidden'; }, 3000); } // --- Fetch Dashboard Stats --- function fetchDashboardStats() { fetch(API_DASHBOARD) .then(res => res.json()) .then(data => { document.getElementById('stat-subjects').textContent = data.total_subjects; document.getElementById('stat-sessions').textContent = data.total_sessions; document.getElementById('stat-completed').textContent = data.completed_sessions; }); } // --- Subject Operations --- function fetchSubjects() { fetch(API_SUBJECTS) .then(res => res.json()) .then(data => { // Populate List subjectList.innerHTML = ''; subjectSelect.innerHTML = ''; data.forEach(sub => { // List Item const li = document.createElement('li'); li.innerHTML = ` ${sub.name} (${sub.duration} mins) `; subjectList.appendChild(li); // Select Option const option = document.createElement('option'); option.value = sub.id; option.textContent = sub.name; subjectSelect.appendChild(option); }); }); } subjectForm.addEventListener('submit', (e) => { e.preventDefault(); const name = document.getElementById('sub-name').value; const duration = document.getElementById('sub-duration').value; fetch(API_SUBJECTS, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name, duration }) }) .then(res => res.json().then(data => ({ status: res.status, body: data }))) .then(res => { if (res.status === 201) { showToast(res.body.message, 'success'); subjectForm.reset(); refreshAll(); } else { showToast(res.body.error, 'error'); } }); }); window.deleteSubject = function(id) { if (!confirm('Are you sure? This will remove scheduled sessions for this subject too.')) return; fetch(`${API_SUBJECTS}/${id}`, { method: 'DELETE' }) .then(res => res.json()) .then(data => { showToast(data.message, 'success'); refreshAll(); }); }; // --- Timetable Operations --- function fetchTimetable() { fetch(API_TIMETABLE) .then(res => res.json()) .then(data => { timetableRows.innerHTML = ''; if (data.length === 0) { timetableRows.innerHTML = `No study sessions scheduled yet.`; return; } data.forEach(entry => { const tr = document.createElement('tr'); if (entry.is_completed) tr.classList.add('completed-row'); tr.innerHTML = ` ${entry.day} ${entry.subject_name} ${entry.start_time} - ${entry.end_time} `; timetableRows.appendChild(tr); }); }); } scheduleForm.addEventListener('submit', (e) => { e.preventDefault(); const editId = editIdInput.value; const payload = { day: document.getElementById('sched-day').value, subject_id: document.getElementById('sched-subject').value, start_time: document.getElementById('sched-start').value, end_time: document.getElementById('sched-end').value }; const url = editId ? `${API_TIMETABLE}/${editId}` : API_TIMETABLE; const method = editId ? 'PUT' : 'POST'; fetch(url, { method: method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }) .then(res => res.json().then(data => ({ status: res.status, body: data }))) .then(res => { if (res.status === 200 || res.status === 201) { showToast(res.body.message, 'success'); resetScheduleForm(); refreshAll(); } else { showToast(res.body.error, 'error'); } }); }); window.toggleComplete = function(id, isCompleted) { fetch(`${API_TIMETABLE}/${id}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ is_completed: isCompleted ? 1 : 0 }) }) .then(() => fetchDashboardStats()); }; window.editSession = function(entry) { editIdInput.value = entry.id; document.getElementById('sched-day').value = entry.day; document.getElementById('sched-subject').value = entry.subject_id; document.getElementById('sched-start').value = entry.start_time; document.getElementById('sched-end').value = entry.end_time; submitSchedBtn.innerHTML = ' Update Session'; cancelEditBtn.classList.remove('hidden'); }; cancelEditBtn.addEventListener('click', resetScheduleForm); function resetScheduleForm() { editIdInput.value = ''; scheduleForm.reset(); submitSchedBtn.innerHTML = ' Add Session'; cancelEditBtn.classList.add('hidden'); } window.deleteSession = function(id) { if (!confirm('Remove this session from your timetable?')) return; fetch(`${API_TIMETABLE}/${id}`, { method: 'DELETE' }) .then(res => res.json()) .then(data => { showToast(data.message, 'success'); refreshAll(); }); }; }); ``` --- ## 🚀 How to Run the Project Locally ### Step 1: Install Dependencies Make sure Python 3.x is installed on your computer.
